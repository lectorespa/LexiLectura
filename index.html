document.addEventListener('DOMContentLoaded', () => { const jsonSelect = document.getElementById('json-select'); const jsonTextarea = document.getElementById('json-input'); const btnLoad = document.getElementById('btn-load-json'); const fileInputGroup = document.getElementById('file-input-group'); const isLocal = window.location.protocol === 'file:'; if (fileInputGroup) fileInputGroup.classList.add('hidden'); let localFilesByName = {}; if (isLocal) { const selectorContainer = document.querySelector('.selector-container'); const folderBtn = document.createElement('input'); folderBtn.type = 'file'; folderBtn.webkitdirectory = true; folderBtn.id = 'folder-picker'; const label = document.createElement('label'); label.innerHTML = ' 📁 Cargar carpeta local: '; selectorContainer.appendChild(label); selectorContainer.appendChild(folderBtn); folderBtn.addEventListener('change', (e) => { const files = e.target.files; localFilesByName = {}; for (let file of files) { const baseName = file.name.split(/[/\\]/).pop().trim(); const cleanName = baseName.toLowerCase(); if (cleanName.endsWith('.json')) { localFilesByName[cleanName] = file; } } obtenerCatalogo(); }); } function obtenerCatalogo() { if (isLocal) return; // Lee el catalogo.json que genera GitHub Actions automáticamente fetch(`./catalogo.json?v=${Date.now()}`, { cache: 'no-store' }) .then(res => { if (!res.ok) throw new Error(`HTTP ${res.status}`); return res.json(); }) .then(data => poblarDesplegable(data)) .catch(err => { console.error('Error al cargar catalogo.json:', err); jsonSelect.innerHTML = ''; }); } function poblarDesplegable(list) { jsonSelect.innerHTML = ''; if (!Array.isArray(list) || list.length === 0) { jsonSelect.innerHTML = ''; return; } list.forEach(item => { const opt = document.createElement('option'); opt.value = item.file.toLowerCase(); opt.textContent = item.author ? `${item.title} — ${item.author}` : item.title; jsonSelect.appendChild(opt); }); if (jsonSelect.options.length > 0) { jsonSelect.selectedIndex = 0; cargarArchivoSeleccionado(); } } function cargarArchivoSeleccionado() { const selectedKey = jsonSelect.value ? jsonSelect.value.toLowerCase() : ''; if (!selectedKey) return; if (isLocal) { const file = localFilesByName[selectedKey]; if (file) { const reader = new FileReader(); reader.onload = (e) => { jsonTextarea.value = e.target.result; if (btnLoad) btnLoad.click(); }; reader.readAsText(file); } } else { const cleanFileName = selectedKey.replace(/^textos\//i, ''); // Apunta a la carpeta /textos/ fetch(`./textos/${cleanFileName}?v=${Date.now()}`, { cache: 'no-store' }) .then(res => { if (!res.ok) throw new Error(`HTTP ${res.status}`); return res.text(); }) .then(textData => { jsonTextarea.value = textData; if (btnLoad) btnLoad.click(); }) .catch(err => console.error(`Error al obtener ./textos/${cleanFileName}:`, err)); } } jsonSelect.addEventListener('change', cargarArchivoSeleccionado); // Inicializar obtenerCatalogo(); });
